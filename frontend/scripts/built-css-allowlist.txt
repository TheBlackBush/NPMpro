# What the built stylesheet may have outside a cascade layer (it beats every layered rule, Tabler's included) and
# which unprefixed Tailwind-looking classes the new UI's layers may style. scripts/check-built-css.js compares this
# list with the build after `vite build`; a new line fails CI, and so does a line the build no longer has.
# Before adding a line: check that it cannot change an old page (a Tabler or Bootstrap class or keyframes name, an
# element selector, a custom property Tabler reads). Tailwind's own @property --tw-* are not listed. Outside Tabler's
# layer and outside @scope every rule must name a tw: class; layered @keyframes, @font-face and @property are findings.
# Format: one finding per line as the check prints it, a comment above each group with its source and why it is safe.

# shadcn/tailwind.css (shadcn package): custom properties of its scroll-fade and shimmer utilities, names Tabler
# does not use
unlayered: @property --scroll-fade-t
unlayered: @property --scroll-fade-b
unlayered: @property --scroll-fade-s
unlayered: @property --scroll-fade-e
unlayered: @property --scroll-fade-mask
unlayered: @property --shimmer-angle
unlayered: @property --shimmer-image
unlayered: @property --shimmer-text-fill
# shadcn/tailwind.css: turns its .shimmer animation off for reduced motion; Tabler has no .shimmer class
unlayered: @media (prefers-reduced-motion:reduce) > .shimmer

# styles/tailwind.css: shadcn's design tokens (--background, --primary, ...) and their dark values; Tabler reads only
# --tblr-* and --bs-* properties
unlayered: :root
unlayered: .dark

# Tailwind's `properties` layer (inside its @supports fallback): initial values of the custom properties the utilities
# in use register, all --tw-* plus shadcn's --scroll-fade-*, --shimmer-* and --shimmer-text-* (names Tabler does not
# use); no element property is set
unprefixed: properties: :root,:host
unprefixed: properties: *,:before,:after,::backdrop

# Tailwind's `theme` layer: the design tokens it generates as --tw-* custom properties (--tw-font-sans, ...); Tabler
# reads only --tblr-* and --bs-*
unprefixed: theme: :root,:host

# Tailwind and tw-animate-css: keyframes of tw:animate-pulse (the skeleton), tw:animate-spin (the command palette's
# loading row) and the enter/exit animations, emitted outside the layers. Tabler also has @keyframes pulse (a scale
# bounce), which this one replaces; only Tabler's .icon-pulse and .btn-animate-icon-pulse use it, and the app uses
# neither. Tabler has no @keyframes spin (its spinners use spinner-border and spinner-grow), so spin replaces nothing
unlayered: @keyframes pulse
unlayered: @keyframes spin
unlayered: @keyframes enter
unlayered: @keyframes exit
